! Tabla de Direccionamiento:

! Dispositivo | Interfaz | IP               | Descripcion
! -------------------------------------------------------------------
! ISP (Hub)   | e0/0     | 20.25.1.1/30     | WAN hacia R1-S1
! ISP (Hub)   | Tunnel0  | 14.2.10.1/24     | DMVPN mGRE Hub
! R1-S1 (Sp1) | e0/0     | 20.25.1.2/30     | WAN hacia ISP
! R1-S1 (Sp1) | e0/2     | 20.25.2.5/30     | WAN hacia R3
! R1-S1 (Sp1) | Tunnel0  | 14.2.10.2/24     | DMVPN GRE Spoke1
! R3 (Sp2)    | e0/0     | 20.25.2.6/30     | WAN hacia R1-S1
! R3 (Sp2)    | e0/1     | 30.30.30.1/24    | LAN
! R3 (Sp2)    | Tunnel0  | 14.2.10.3/24     | DMVPN GRE Spoke2
! -------------------------------------------------------------------
! R3 llega al ISP (Hub) pasando por R1-S1 como transit
! PSK: Cisco123 | NHRP Auth: NHRP2024 | EIGRP AS 100
! Fase 3: Hub envia NHRP Redirect; spokes usan CEF shortcut
! ============================================================


! ============================================================
! ISP - HUB DMVPN FASE 3
! ============================================================
enable
conf t
hostname ISP
!
! --- IKEv2 Proposal ---
crypto ikev2 proposal PROP_IKEv2
 encryption aes-cbc-256
 integrity sha256
 group 14
!
crypto ikev2 policy POL_IKEv2
 proposal PROP_IKEv2
!
crypto ikev2 keyring KEYRING_DMVPN
 peer ANY_SPOKE
  address 0.0.0.0 0.0.0.0
  pre-shared-key Cisco123
!
crypto ikev2 profile PROF_IKEv2
 match identity remote address 0.0.0.0
 authentication remote pre-share
 authentication local pre-share
 keyring local KEYRING_DMVPN
!
crypto ipsec transform-set TS_DMVPN esp-aes 256 esp-sha256-hmac
 mode transport
!
crypto ipsec profile DMVPN_PROFILE
 set transform-set TS_DMVPN
 set ikev2-profile PROF_IKEv2
!
! Diferencia clave Fase 3: ip nhrp redirect en el Hub
interface Tunnel0
 ip address 14.2.10.1 255.255.255.0
 no ip redirects
 ip nhrp authentication NHRP2024
 ip nhrp map multicast dynamic
 ip nhrp network-id 100
 ip nhrp holdtime 600
 ip nhrp redirect
 tunnel source e0/0
 tunnel mode gre multipoint
 tunnel protection ipsec profile DMVPN_PROFILE
!
interface e0/0
 ip address 20.25.1.1 255.255.255.252
 no shutdown
!
router eigrp 100
 network 14.2.10.0 0.0.0.255
 network 20.25.1.0 0.0.0.3
 no auto-summary
!
end


! ============================================================
! R1-S1 - SPOKE 1 (FASE 3 + IKEv2)
! ============================================================
enable
conf t
hostname R1-S1
!
crypto ikev2 proposal PROP_IKEv2
 encryption aes-cbc-256
 integrity sha256
 group 14
!
crypto ikev2 policy POL_IKEv2
 proposal PROP_IKEv2
!
crypto ikev2 keyring KEYRING_DMVPN
 peer HUB
  address 20.25.1.1
  pre-shared-key Cisco123
!
crypto ikev2 profile PROF_IKEv2
 match identity remote address 20.25.1.1 255.255.255.255
 authentication remote pre-share
 authentication local pre-share
 keyring local KEYRING_DMVPN
!
crypto ipsec transform-set TS_DMVPN esp-aes 256 esp-sha256-hmac
 mode transport
!
crypto ipsec profile DMVPN_PROFILE
 set transform-set TS_DMVPN
 set ikev2-profile PROF_IKEv2
!
! Diferencia clave Fase 3: ip nhrp shortcut en los Spokes
interface Tunnel0
 ip address 14.2.10.2 255.255.255.0
 ip nhrp authentication NHRP2024
 ip nhrp map multicast 20.25.1.1
 ip nhrp map 14.2.10.1 20.25.1.1
 ip nhrp network-id 100
 ip nhrp holdtime 600
 ip nhrp nhs 14.2.10.1
 ip nhrp shortcut
 tunnel source e0/0
 tunnel mode gre multipoint
 tunnel protection ipsec profile DMVPN_PROFILE
!
interface e0/0
 ip address 20.25.1.2 255.255.255.252
 no shutdown
!
interface e0/2
 ip address 20.25.2.5 255.255.255.252
 no shutdown
!
router eigrp 100
 network 14.2.10.0 0.0.0.255
 network 20.25.1.0 0.0.0.3
 network 20.25.2.4 0.0.0.3
 no auto-summary
!
end


! ============================================================
! R3 - SPOKE 2 (FASE 3 + IKEv2)
! ============================================================
enable
conf t
hostname R3
!
crypto ikev2 proposal PROP_IKEv2
 encryption aes-cbc-256
 integrity sha256
 group 14
!
crypto ikev2 policy POL_IKEv2
 proposal PROP_IKEv2
!
! R3 hace IKEv2 hacia el Hub ISP (20.25.1.1)
! El GRE viaja via R1-S1 como transit
crypto ikev2 keyring KEYRING_DMVPN
 peer HUB
  address 20.25.1.1
  pre-shared-key Cisco123
!
crypto ikev2 profile PROF_IKEv2
 match identity remote address 20.25.1.1 255.255.255.255
 authentication remote pre-share
 authentication local pre-share
 keyring local KEYRING_DMVPN
!
crypto ipsec transform-set TS_DMVPN esp-aes 256 esp-sha256-hmac
 mode transport
!
crypto ipsec profile DMVPN_PROFILE
 set transform-set TS_DMVPN
 set ikev2-profile PROF_IKEv2
!
interface Tunnel0
 ip address 14.2.10.3 255.255.255.0
 ip nhrp authentication NHRP2024
 ip nhrp map multicast 20.25.1.1
 ip nhrp map 14.2.10.1 20.25.1.1
 ip nhrp network-id 100
 ip nhrp holdtime 600
 ip nhrp nhs 14.2.10.1
 ip nhrp shortcut
 tunnel source e0/0
 tunnel mode gre multipoint
 tunnel protection ipsec profile DMVPN_PROFILE
!
interface e0/0
 ip address 20.25.2.6 255.255.255.252
 no shutdown
!
interface e0/1
 ip address 30.30.30.1 255.255.255.0
 no shutdown
!
ip route 20.25.1.0 255.255.255.252 20.25.2.5
ip route 14.2.10.0 255.255.255.0 20.25.2.5
!
router eigrp 100
 network 14.2.10.0 0.0.0.255
 network 20.25.2.4 0.0.0.3
 network 30.30.30.0 0.0.0.255
 no auto-summary
!
end
